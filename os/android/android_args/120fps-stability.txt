# 120 Hz / 120 FPS stability baseline for high-end Adreno devices.
# Applies no deliberate quality reductions relative to the Android default.
# TEST PROFILE; copy to args.txt and retain a backup of your current settings.
# Not intended for Snapdragon 865 or 60 Hz displays.

--gta4_draw_distance_scale=1.0
--gta4_shadow_map_base_size=256

--gta4_parked_car_density_scale=0.4
--gta4_ped_density_scale=0.4
--gta4_scenario_ped_density_scale=0.4
--gta4_profile_native_detailed_cpu=false
--gta4_profile_native_detailed_gpu=false
--gta4_output_height_cap=720
